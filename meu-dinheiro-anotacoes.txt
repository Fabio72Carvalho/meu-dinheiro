import { salvarConta } from './db.js';

// Elementos da seção contas
const secaoContas = document.getElementById('secao-contas');
const btnSalvarConta = document.getElementById('btn-salvar-conta');
const inputNomeConta = document.getElementById('nome-conta');
const inputSaldoInicial = document.getElementById('saldo-inicial');


// --- EVENTOS DE BANCO DE DADOS (Firestore) ---
if (btnSalvarConta) {
    btnSalvarConta.addEventListener('click', async () => {
        const nome = inputNomeConta.value;
        const saldo = inputSaldoInicial.value;

        if (!nome || !saldo) {
            alert("Por favor, preencha todos os campos da conta.");
            return;
        }

        if (usuarioAtual) {
            try {
                const dadosConta = {
                    nome: nome,
                    saldoAtual: parseFloat(saldo)
                };

                await salvarConta(usuarioAtual.uid, dadosConta);

                alert("Conta '" + nome + "' cadastrada com sucesso!");

                // Limpar campos
                inputNomeConta.value = '';
                inputSaldoInicial.value = '';

            } catch (error) {
                alert("Erro ao salvar conta: " + error.message);
            }
        } else {
            alert("Você precisa estar logado para cadastrar uma conta.");
        }
    });
}

🚀 Ticket de Continuidade: Projeto "Meu Dinheiro"
Status Atual (O que já temos rodando liso):

Motor Mestre (Firestore): Transações, receitas, despesas e transferências estão sendo salvas via writeBatch. O saldo das contas e os consolidados mensais (saldos_anuais) são atualizados de forma atômica e perfeita.

Interface (UI): O ui.js foi unificado. A barra lateral lê o saldo atualizado dinamicamente. O extrato central retro-calcula o saldo inicial do mês e desenha o extrato corrido com precisão contábil.

Filtros Mágicos: O "Maestro" (aplicarFiltrosMemoria) no app.js cruza dados de contas, categorias e transações instantaneamente.

Onde Paramos:
